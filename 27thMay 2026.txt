// $in matching exact values & regular expressions
db.products.find({ price: { $in: [699, 799] } }, { name: 1, price: 1 });
db.products.find({ color: { $in: ["black", "white"] } }, { name: 1, color: 1 });
db.products.find({ color: { $in: [/^g+/, /^w+/] } }, { name: 1, color: 1 });

// $nin (Not In) operations
db.products.find({ price: { $nin: [699, 799] } }, { name: 1, price: 1 });
db.products.find({ color: { $nin: ["black", "white"] } }, { name: 1, color: 1 });
db.products.find({ color: { $nin: [/^g+/, /^w+/] } }, { name: 1, color: 1 });


// 5. Logical Operators Setup ($and, $or, $not, $nor)

// Evaluating Logical AND ($and) conditions
db.products.find({ $and: [{ price: 899 }, { color: { $in: ["white", "black"] } }] }, { name: 1, price: 1, color: 1 });
db.products.find({ $and: [{ price: 699 }, { price: { $exists: true } }] }, { name: 1, price: 1, color: 1 });

// Explicit vs Implicit equality on field matching
db.products.find({ price: { $eq: 899 } }, { name: 1, price: 1 });
db.products.find({ "spec.ram": 4 }, { name: 1, "spec.ram": 1 });

// Evaluating Logical OR ($or) conditions
db.products.find({ $or: [{ price: 799 }, { price: 899 }] }, { name: 1, price: 1 });
db.products.find({ $or: [{ price: { $lt: 699 } }, { price: { $gt: 799 } }] }, { name: 1, price: 1 });

// Evaluating Logical NOT ($not) conditions
db.products.find({ price: { $not: { $gt: 699 } } }, { name: 1, price: 1 });
db.products.find({ name: { $not: /^Smart+/ } }, { name: 1 });

// Evaluating Logical NOR ($nor) conditions
db.products.find({ $nor: [{ price: 899 }, { color: "gold" }] }, { name: 1, price: 1, color: 1 });
