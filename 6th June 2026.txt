// Attempting unique creation directly on an existing indexed key causing IndexKeySpecsConflict error

// 4. Data Integrity Auditing via Single-Field Unique Constraints

   { email: "john@test.com", name: "john" },
   { email: "jane@test.com", name: "jane" }
]);

// Testing data validation safety (Triggers E11000 Duplicate Key Protection Engine)
// db.users.insertOne({ email: "john@test.com", name: "johny" }); 

// Handling DuplicateKey exception on live collection configurations
db.users.drop();
db.users.insertMany([
   { email: "john@test.com", name: "john" },
   { email: "john@test.com", name: "johny" }, // Pre-existing duplicate element block
   { email: "jane@test.com", name: "jane" }
]);

// Attempting unique index assignment here throws DuplicateKey error until records are resolved
db.users.deleteOne({ name: 'johny', email: 'john@test.com' }); // Resolving redundancy row context
db.users.createIndex({ email: 1 }, { unique: true }); // Successful deployment validation
