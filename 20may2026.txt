-- ==========================================
-- [FIRSTQUERY] - Unique & Constraints
-- ==========================================
USE VITCampus;

-- Drop table if exists to avoid collision
DROP TABLE IF EXISTS Persons;

CREATE TABLE Persons (
    ID INT NOT NULL,
    LastName VARCHAR(255) UNIQUE NOT NULL,
    FirstName VARCHAR(255) NOT NULL,
    Age INT
);

DESC Persons;

-- Inserting sample data with NULL age
INSERT INTO Persons VALUES (101, 'V', 'Jayanth', NULL);
SELECT * FROM Persons;

-- Adding and dropping constraints
ALTER TABLE Persons ADD UNIQUE (Age);
ALTER TABLE Persons DROP INDEX Age; 


-- ==========================================
-- [SECONDQUERY] - Check Constraints
-- ==========================================
DROP TABLE IF EXISTS Persons;

CREATE TABLE Persons (
    ID INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Age INT,
    CHECK (Age >= 18)
);

-- Category Table Setup (Parent Table)
DROP TABLE IF EXISTS category;
CREATE TABLE category (
    cid INT PRIMARY KEY,
    cname VARCHAR(100) NOT NULL
);

INSERT INTO category VALUES (101, 'electronics'), (102, 'Home App');
SELECT * FROM category;

-- Products Table Setup (Child Table with Foreign Key)
DROP TABLE IF EXISTS products;
CREATE TABLE products (
    pid INT PRIMARY KEY,
    pname VARCHAR(100) NOT NULL,
    cid INT, 
    FOREIGN KEY (cid) REFERENCES category(cid)
);

DESC products;
