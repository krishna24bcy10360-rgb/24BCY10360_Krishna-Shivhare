// 2. Type Constraints Auditing using $type (String representations or BSON Alias Type IDs)
db.products.find({ price: { $type: "string" } }, { name: 1, price: 1 });
db.products.find({ price: { $type: 2 } }, { name: 1, price: 1 }); // 2 maps to Type String
db.products.find({ price: { $type: "number" } }, { name: 1, price: 1 });
db.products.find({ price: { $type: "array" } }, { name: 1, price: 1 });
db.products.find({ price: { $type: ["number", "string"] } }, { name: 1, price: 1 });
db.products.find({ releaseDate: { $type: "date" } }, { name: 1, price: 1 });


// 3. Existence Checks ($exists)

db.products.find({ price: { $exists: false } }, { name: 1, price: 1 });
