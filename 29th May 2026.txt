// 2. Advanced Multi-Key Sorting and Filtering
db.products.find({ 'price': { $exists: 1 } }, { name: 1, price: 1 });
db.products.find({ 'price': { $exists: 1 } }, { name: 1, price: 1 }).sort({ price: 1 });
db.products.find({ 'price': { $exists: 1 } }, { name: 1, price: 1 }).sort({ price: -1 });

// Sorting by price (ascending) and resolving ties with name (ascending)
db.products.find({ 'price': { $exists: 1 } }, { name: 1, price: 1 }).sort({ price: 1, name: 1 });

// Sorting by price (ascending) and resolving ties with name (descending)
db.products.find({ 'price': { $exists: 1 } }, { name: 1, price: 1 }).sort({ price: 1, name: -1 });


// 3. Date and Embedded Document Object Sorting

db.products.find({}, { name: 1, spec: 1 }).sort({ "spec.ram": 1 });


// 4. Pagination Techniques Using Limit & Skip
db.products.find({}, { name: 1, price: 1 }).sort({ price: -1 }).limit(1);
db.products.find({}, { name: 1, price: 1 }).sort({ price: -1, name: 1 }).limit(1);

// Skipping first 2 matched data slots and evaluating next 2 segments
db.products.find({}, { name: 1, price: 1 }).sort({ price: -1, name: 1 }).skip(2).limit(2);
