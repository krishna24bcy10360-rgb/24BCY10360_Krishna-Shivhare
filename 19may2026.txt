-- ==========================================
-- [FIRSTQUERY] - Basic Filtering & Operators
-- ==========================================
USE VITCampus;

SHOW TABLES FROM VITCampus;

SELECT * FROM worker;

-- Filtering with OR and Comparison
SELECT first_name, department, salary FROM worker
WHERE department='Admin' OR salary >= 100000;

-- Filtering with NOT IN
SELECT worker_id, first_name, department, salary FROM worker
WHERE worker_id NOT IN (1, 6, 7);

-- Filtering with NOT BETWEEN
SELECT * FROM worker 
WHERE salary NOT BETWEEN 110000 AND 300000;

-- Combining BETWEEN and NOT IN
SELECT * FROM worker
WHERE salary BETWEEN 100000 AND 500000
AND department NOT IN ('HR', 'Admin');

-- ==========================================
-- [THIRDQUERY] - Pattern Matching, Case & Union
-- ==========================================
SELECT * FROM worker;

-- Pattern Matching with LIKE
SELECT * FROM worker WHERE first_name LIKE '_i__a_';

SELECT (salary) FROM worker WHERE department='Admin';

-- UNION ALL Demonstration
SELECT salary FROM worker
UNION ALL
SELECT salary FROM worker;

-- Column Aliasing
SELECT worker_id AS RollNumber FROM worker;

-- Distinct Departments
SELECT DISTINCT department FROM worker;

-- Conditional CASE Statement
SELECT worker_id, first_name,
CASE
    WHEN salary > 300000 THEN 'Rich people'
    WHEN salary >= 100000 AND salary < 299999 THEN 'Middle'
    ELSE 'Poor'
END AS Status
FROM worker;


-- ==========================================
-- [FOURTHQUERY] - Aggregations and SQL Views
-- ==========================================
SELECT MIN(salary) FROM worker WHERE department='Admin';
SELECT MAX(salary) FROM worker WHERE department='Admin';

-- Creating Views safely using OR REPLACE
CREATE OR REPLACE VIEW minSalaryAdmin AS 
SELECT MIN(salary) AS Min_Salary FROM worker WHERE department='Admin';

CREATE OR REPLACE VIEW mAxSalaryAdmin AS 
SELECT MAX(salary) AS Max_Salary FROM worker WHERE department='Admin';

-- Verifying the View
